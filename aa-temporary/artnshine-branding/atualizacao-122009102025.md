### **A. Criar `scripts/optimize-images.js`:**
```javascript
// Otimização automática de imagens
const sharp = require('sharp');
const fs = require('fs');
const path = require('path');

async function optimizeProductImages() {
    const inputDir = 'public/uploads/products/';
    const files = fs.readdirSync(inputDir);
    
    for (const file of files) {
        if (file.match(/\.(jpg|jpeg)$/i)) {
            await sharp(path.join(inputDir, file))
                .resize(800, 1000, { 
                    fit: 'cover',
                    position: 'center'
                })
                .jpeg({ quality: 85 })
                .toFile(path.join(inputDir, `opt_${file}`));
                
            console.log(`Otimized: ${file}`);
        }
    }
}

async function optimizeHeroBgs() {
    const inputDir = 'public/images/backgrounds/';
    const files = fs.readdirSync(inputDir);
    
    for (const file of files) {
        if (file.match(/\.(jpg|jpeg)$/i)) {
            await sharp(path.join(inputDir, file))
                .resize(1920, 1080, { 
                    fit: 'cover',
                    position: 'center'
                })
                .jpeg({ quality: 80 })
                .toFile(path.join(inputDir, `opt_${file}`));
                
            console.log(`Hero optimized: ${file}`);
        }
    }
}

// Executar
optimizeProductImages().then(() => {
    return optimizeHeroBgs();
}).then(() => {
    console.log('All images optimized!');
});
```
